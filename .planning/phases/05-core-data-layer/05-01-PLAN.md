class FieldSchema {
  const FieldSchema({
    required this.key,
    required this.label,
    required this.inputType,
    this.required = false,
    this.hint,
    this.options,
    this.min,
    this.max,
    this.pattern,
    this.enumOptions,
    this.derivedFrom,
    this.defaultValue,
  });

  final String key;
  final String label;
  final FieldInputType inputType;
  final bool required;
  final String? hint;
  final List<String>? options;          // for select input type
  final num? min;                        // for number input type
  final num? max;                        // for number input type
  final String? pattern;                 // regex for text input type
  final List<String>? enumOptions;       // explicit enum values
  final String? derivedFrom;             // formula string like "POW * 5" (D-03)
  final dynamic defaultValue;

  Map<String, dynamic> toJson() => {
    'key': key,
    'label': label,
    'inputType': inputType.name,
    'required': required,
    if (hint != null) 'hint': hint,
    if (options != null) 'options': options,
    if (min != null) 'min': min,
    if (max != null) 'max': max,
    if (pattern != null) 'pattern': pattern,
    if (enumOptions != null) 'enumOptions': enumOptions,
    if (derivedFrom != null) 'derivedFrom': derivedFrom,
    if (defaultValue != null) 'defaultValue': defaultValue,
  };

  factory FieldSchema.fromJson(Map<String, dynamic> json) => FieldSchema(
    key: json['key'] as String,
    label: json['label'] as String,
    inputType: FieldInputType.values.byName(json['inputType'] as String),
    required: json['required'] as bool? ?? false,
    hint: json['hint'] as String?,
    options: (json['options'] as List<dynamic>?)?.map((e) => e as String).toList(),
    min: json['min'] as num?,
    max: json['max'] as num?,
    pattern: json['pattern'] as String?,
    enumOptions: (json['enumOptions'] as List<dynamic>?)?.map((e) => e as String).toList(),
    derivedFrom: json['derivedFrom'] as String?,
    defaultValue: json['defaultValue'],
  );
}
```

Follow conventions: `snake_case.dart` file name, `PascalCase` class names, `const` constructor (all fields are final), named `required` parameters, default values for optional fields, manual `toJson`/`fromJson` serialization pattern matching value_types.dart.

class EntityTypeSchema {
  const EntityTypeSchema({
    required this.key,
    required this.displayName,
    required this.isWikiPageType,
    required this.fields,
    this.description,
    this.iconKey,
    this.sortOrder = 0,
  });

  final String key;
  final String displayName;
  final bool isWikiPageType;
  final List<FieldSchema> fields;
  final String? description;
  final String? iconKey;
  final int sortOrder;

  Map<String, dynamic> toJson() => {
    'key': key,
    'displayName': displayName,
    'isWikiPageType': isWikiPageType,
    'fields': fields.map((f) => f.toJson()).toList(),
    if (description != null) 'description': description,
    if (iconKey != null) 'iconKey': iconKey,
    'sortOrder': sortOrder,
  };

  factory EntityTypeSchema.fromJson(Map<String, dynamic> json) => EntityTypeSchema(

    isWikiPageType: json['isWikiPageType'] as bool,

import 'entity_type_schema.dart';

class GameModel {
  const GameModel({
    required this.schemaVersion,
    required this.name,
    required this.entityTypes,
    required this.rulesConfig,
  });

  final int schemaVersion;
  final String name;
  final List<EntityTypeSchema> entityTypes;
  final Map<String, dynamic> rulesConfig;

  Map<String, dynamic> toJson() => {
    'schemaVersion': schemaVersion,
    'name': name,
    'entityTypes': entityTypes.map((e) => e.toJson()).toList(),
    'rulesConfig': rulesConfig,
  };

  factory GameModel.fromJson(Map<String, dynamic> json) => GameModel(
    schemaVersion: json['schemaVersion'] as int,
    name: json['name'] as String,
    entityTypes: (json['entityTypes'] as List<dynamic>)
        .map((e) => EntityTypeSchema.fromJson(Map<String, dynamic>.from(e as Map)))
        .toList(),
    rulesConfig: Map<String, dynamic>.from(json['rulesConfig'] as Map),
  );
}
```

- `const` constructor — all fields are final
