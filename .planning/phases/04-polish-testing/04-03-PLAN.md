**Group 1: 'Responsive breakpoints'** — Test modal panel structure at boundary widths per D-07/D-08:

```dart
import 'package:companion_app/main.dart';
import 'package:core/wiki/wiki.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';

Future<void> openWikiModal(WidgetTester tester) async {
  await tester.pumpWidget(const CompanionApp());
  await tester.tap(find.byIcon(Icons.menu_book));
  await tester.pumpAndSettle();
}

void main() {
  group('Companion wiki modal responsive breakpoints', () {
    testWidgets('shows single panel at 599dp (phone)', (tester) async {
      tester.view.physicalSize = const Size(599, 800);
      tester.view.devicePixelRatio = 1.0;
      addTearDown(tester.view.resetPhysicalSize);

      await openWikiModal(tester);

      // Single panel: only WikiPageList visible, no Row with two children
      expect(find.byType(WikiPageList), findsOneWidget);
      // The body should NOT be a Row (two-panel uses Row)
      final scaffold = tester.widget<Scaffold>(find.byType(Scaffold));
      final body = scaffold.body;
      expect(body is Row, isFalse);
    });

    testWidgets('shows two-panel at 600dp (tablet)', (tester) async {
      tester.view.physicalSize = const Size(600, 800);
      tester.view.devicePixelRatio = 1.0;
      addTearDown(tester.view.resetPhysicalSize);

      await openWikiModal(tester);

      // Two-panel: body is a Row with sidebar (SizedBox width 300) + detail
      final scaffold = tester.widget<Scaffold>(find.byType(Scaffold));
      final body = scaffold.body;
      expect(body is Row, isTrue);
      final row = body as Row;
      expect(row.children.length, equals(2));
      // First child is SizedBox with width 300 (sidebar)
      final sidebar = row.children.first;
      expect(sidebar is SizedBox, isTrue);
      expect((sidebar as SizedBox).width, equals(300));
    });

    testWidgets('shows two-panel at 840dp (desktop lower boundary)', (tester) async {
      tester.view.physicalSize = const Size(840, 800);
      tester.view.devicePixelRatio = 1.0;
      addTearDown(tester.view.resetPhysicalSize);

      await openWikiModal(tester);

      final scaffold = tester.widget<Scaffold>(find.byType(Scaffold));
      expect(scaffold.body is Row, isTrue);
    });

    testWidgets('shows two-panel at 841dp (desktop upper boundary)', (tester) async {
      tester.view.physicalSize = const Size(841, 800);
      tester.view.devicePixelRatio = 1.0;
      addTearDown(tester.view.resetPhysicalSize);

      await openWikiModal(tester);

      final scaffold = tester.widget<Scaffold>(find.byType(Scaffold));
      expect(scaffold.body is Row, isTrue);
    });
  });
```

      final textField = tester.widget<TextField>(find.byType(TextField));

    testWidgets('search input filters page list', (tester) async {
      // This test verifies the search TextField is wired to filtering.
      // The debounce timing is tested at the unit level (wiki_debounce_test.dart).
      await openWikiModal(tester);

      final searchField = find.byType(TextField);
      expect(searchField, findsOneWidget);

      // Enter text — the list should filter (debounce handled by DebounceUtil)
      await tester.enterText(searchField, 'nonexistent-page-xyz');
      await tester.pump(const Duration(milliseconds: 300)); // Wait for 250ms debounce + buffer
      await tester.pump();

      // ListView should show 0 items (no matching pages in fresh app)
      final listView = find.byType(ListView);
      expect(listView, findsOneWidget);
    });
  });
}
